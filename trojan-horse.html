function handleClick(e) { if (e.__thDone) return; e.__thDone = true; try { const act = closestOf(e.target, '[data-action]'); if (act) return handleAction(act.dataset.action, act); const cm = closestOf(e.target, '.common-chip'); if (cm) return toggleCommon(cm.dataset.common); const pill = closestOf(e.target, '.pill'); if (pill) return togglePill(pill); const chip = closestOf(e.target, '.chip'); if (chip) return toggleChip(chip); const lab = closestOf(e.target, '.play-label'); if (lab) return toggleDesc(lab); const lad = closestOf(e.target, '.ladder-row'); if (lad) return toggleLadder(lad); const tc = closestOf(e.target, '.tone-chip'); if (tc) { state.tone = state.tone === tc.dataset.val ? '' : tc.dataset.val; return paint(); } const sw = closestOf(e.target, '.switch'); if (sw) return toggleSwitch(sw); } catch (err) { fatal(err); } } const CLICKABLE = '[data-action],.chip,.pill,.common-chip,.tone-chip,.switch,.ladder-row,.play-label'; function bindInteractive(root) { const scope = root || document; let n = 0; if (scope.querySelectorAll) { const list = scope.querySelectorAll(CLICKABLE); for (let i = 0; i < list.length; i++) { const el = list[i]; if (el.__thBound) continue; el.__thBound = true; el.addEventListener('click', handleClick); n++; } } if (scope.nodeType === 1 && !scope.__thBound && typeof scope.matches === 'function' && scope.matches(CLICKABLE)) { scope.__thBound = true; scope.addEventListener('click', handleClick); n++; } return n; } /* 两条兜底通道在脚本刚跑到这里时就挂上，哪怕后面的初始化炸了也还在。 */ document.addEventListener('click', handleClick); window.addEventListener('click', handleClick, true);
