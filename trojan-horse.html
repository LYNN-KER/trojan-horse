function genLines(c) { const g = c.gen, L = []; if (g.types.includes('none')) { L.push(t(B('生殖器：无。不得描写。', 'Genitalia: none. Do not describe any.'))); return L; } if (!g.types.length) return L; const ty = g.types.map(k => labelOf(G_TYPES, k)).join(' + '); L.push(t(B(`生殖器类型：${ty}`, `Genitalia present: ${ty}`))); if (g.types.includes('multiple')) { L.push(t(B('存在多副生殖结构。每场自行决定哪一副在场，其余不提及。', 'More than one set exists. Decide per scene which one is in play; the rest are simply not mentioned.'))); } if (hasP(c)) { const form = g.pform ? labelOf(OPTS.pform, g.pform) : labelOf(OPTS.pform, 'human'); const bits = []; if (g.plen) bits.push(t(B(`长 ${g.plen}cm`, `${g.plen}cm long`))); if (g.pgirth) bits.push(t(B(`周长 ${g.pgirth}cm`, `${g.pgirth}cm girth`))); L.push(t(B(`阴茎：存在，形态 ${form}。`, `Penis: present — form ${form}.`)) + (bits.length ? t(B(' 已量：', ' Measured: ')) + joinWith(bits, '、', ', ') + t(B('。', '.')) : t(B(' 比例与上述体型一致。', ' Proportions consistent with the physique above.')))); if (g.pstate) { const vis = labelOf(OPTS.pstate, g.pstate, 'pstate'); L.push(t(B(`阴茎显现方式：${vis}。`, `Penis visibility: ${vis}.`))); } L.push(g.pdesc ? t(B(`阴茎外观（照此描写，不得替换成人类版本）：${g.pdesc}`, `Penis appearance (as described; do not substitute a human one): ${g.pdesc}`)) : t(B('阴茎外观：未指定——遵循上面的形态，并在整场中保持解剖一致。', 'Penis appearance: unstated — follow the form above and keep the anatomy consistent throughout.'))); } if (hasV(c)) { const bits = []; if (g.vdepth) bits.push(t(B(`深 ${g.vdepth}cm`, `${g.vdepth}cm deep`))); if (g.vaperture) bits.push(t(B(`入口直径 ${g.vaperture}cm`, `${g.vaperture}cm aperture`))); L.push(t(B('阴道：存在。', 'Vagina: present.')) + (bits.length ? ' ' + joinWith(bits, '、', ', ') + t(B('。', '.')) : t(B(' 比例与上述体型一致。', ' Proportions consistent with the physique above.')))); if (g.vtraits.length) { const tr = g.vtraits.map(k => labelOf(OPTS.traits, k)).join('; '); L.push(t(B(`阴道内部特征：${tr}。`, `Vagina traits: ${tr}.`))); } L.push(g.vdesc ? t(B(`阴道外观（照此描写）：${g.vdesc}`, `Vagina appearance (as described): ${g.vdesc}`)) : t(B('阴道外观：未指定——保持解剖一致，场景没要求时不过度细化。', 'Vagina appearance: unstated — keep it anatomically consistent and non-graphic unless the scene calls for more.'))); } if (g.orifices.length) { const orc = g.orifices.map(o => labelOf(OPTS.orifices, o)).join('; '); L.push(t(B(`其他可用的孔位：${orc}。每个只在与该角色解剖相符的姿势下使用。`, `Additional openings available: ${orc}. Use each only in positions anatomically consistent with this character.`))); } return L; } function charLines(c) { const L = []; const nh = isNH(c); L.push(t(B(`种族：${nh ? '非人形' : '人形'}`, `Species: ${nh ? 'Non-human' : 'Human'}`))); if (nh && c.subspecies) { const k = labelOf(OPTS.subspecies, c.subspecies); L.push(t(B(`亚种：${k}`, `Kind: ${k}`))); } if (nh && c.material) { const m = labelOf(OPTS.material, c.material); L.push(t(B(`质地：${m}`, `Material: ${m}`))); } if (c.gender) { const gi = labelOf(OPTS.gender, c.gender); L.push(t(B(`性别认同：${gi}`, `Gender identity: ${gi}`))); } if (c.h || c.w || c.build) { const hs = c.h ? c.h + 'cm' : t(B('身高未指定', 'height unspecified')); const wt = c.w ? ' / ' + c.w + 'kg' : ''; const bd = c.build ? ' / ' + labelOf(OPTS.build, c.build, 'build') : ''; L.push(t(B(`体型：${hs}${wt}${bd}`, `Physique: ${hs}${wt}${bd}`))); } L.push(...genLines(c)); if (nh) { if (c.tent) L.push(t(B(`肢体：${c.tent} 条触手${c.tlen ? '，每条 ' + c.tlen + 'cm' : ''}`, `Limbs: ${c.tent} tentacles${c.tlen ? `, each ${c.tlen}cm` : ''}`))); if (c.tail) { const tl = labelOf(OPTS.tail, c.tail); L.push(t(B(`尾巴：${tl}`, `Tail: ${tl}`))); } if (c.extra) L.push(t(B(`其他特征：${c.extra}`, `Other features: ${c.extra}`))); } return L; }
